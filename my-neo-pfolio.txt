const esc = s => String(s).replace(/[&<>"]/g, c => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[c]));
const initials = ME.name.split(/\s+/).map(w => w[0]).join("").slice(0,2).toUpperCase();

document.title = ME.name + " — Portfolio";
$("#name").textContent = ME.name; $("#logoName").textContent = initials; $("#tagline").textContent = ME.tagline;
$("#bio").textContent = ME.bio; $("#foot").textContent = "© " + new Date().getFullYear() + " " + ME.name + ". All rights reserved.";
if (ME.photo) $("#avatar").innerHTML = '<img alt="' + esc(ME.name) + '" src="' + esc(ME.photo) + '">'; else $("#initials").textContent = initials;

$("#stats").innerHTML = ME.stats.map(s => '<div class="stat"><strong class="grad">' + esc(s[0]) + '</strong><small>' + esc(s[1]) + '</small></div>').join("");

$("#chips").innerHTML = ME.extras.map(e => '<span class="chip">' + esc(e) + '</span>').join("");
$("#timeline").innerHTML = ME.journey.map(j => '<div class="tl-item"><div class="tl-date">' + esc(j[0]) + '</div><div><h3>' + esc(j[1]) + '</h3><p>' + esc(j[2]) + '</p></div></div>').join("");

const c = ME.contact, items = [];
if (c.email) items.push(["✉", c.email, "mailto:" + c.email]);
if (c.phone) items.push(["☎", c.phone, "tel:" + c.phone]);
if (c.location) items.push(["⌖", c.location, "#contact"]);
if (c.github) items.push(["⌘", "GitHub", c.github]);
if (c.linkedin) items.push(["in", "LinkedIn", c.linkedin]);
$("#contactList").innerHTML = items.map(i => '<a href="' + esc(i[2]) + '" target="_blank" rel="noopener"><span class="grad" style="font-weight:700;width:24px;text-align:center">' + i[0] + '</span>' + esc(i[1]) + '</a>').join("");

(function tick(){
  const word = ME.roles[ri];
  $("#typed").textContent = word.slice(0, ci);
  if (!del && ci === word.length) { del = true; return setTimeout(tick, 1400); }
  if (del && ci === 0) { del = false; ri = (ri + 1) % ME.roles.length; }
  ci += del ? -1 : 1;
  setTimeout(tick, del ? 40 : 85);

  io.unobserve(e.target);
}), {threshold:.15});
document.querySelectorAll(".reveal").forEach(el => io.observe(el));

  links.forEach(a => a.classList.toggle("active", a.getAttribute("href") === "#" + cur));
}, {passive:true});

  try { localStorage.setItem("theme", root.dataset.theme); } catch (e) {}
